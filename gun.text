
-- TELEPORT TESTER - UPDATED
-- Place in StarterPlayerScripts (Roblox Studio)

local Players = game:GetService("Players")
local UIS = game:GetService("UserInputService")

local player = Players.LocalPlayer
local playerGui = player:WaitForChild("PlayerGui")

local GUI_NAME = "TeleportTester"
local STOP_NAME = "_TeleportTester_Stop"
local GENERATION_NAME = "_TeleportTester_Generation"

--==================================================
-- VERSION CONTROL: stop previous script instance
--==================================================

local generation = playerGui:FindFirstChild(GENERATION_NAME)

if not generation then
	generation = Instance.new("IntValue")
	generation.Name = GENERATION_NAME
	generation.Value = 0
	generation.Parent = playerGui
end

generation.Value += 1
local myGeneration = generation.Value

local stopEvent = playerGui:FindFirstChild(STOP_NAME)

if not stopEvent then
	stopEvent = Instance.new("BindableEvent")
	stopEvent.Name = STOP_NAME
	stopEvent.Parent = playerGui
end

stopEvent:Fire()

local connections = {}
local cleaned = false
local gui

local function track(connection)
	table.insert(connections, connection)
	return connection
end

local function isCurrent()
	return not cleaned
		and generation.Parent == playerGui
		and generation.Value == myGeneration
end

local function cleanup()
	if cleaned then return end
	cleaned = true

	for _, connection in ipairs(connections) do
		if connection.Connected then
			connection:Disconnect()
		end
	end

	table.clear(connections)

	if gui and gui.Parent then
		gui:Destroy()
	end
end

track(stopEvent.Event:Connect(cleanup))

local oldGui = playerGui:FindFirstChild(GUI_NAME)
if oldGui then
	oldGui:Destroy()
end

--==================================================
-- SETTINGS
--==================================================

local allowedTools = {
	["Greatclub"] = true,
	["Bronze Sword"] = true,
	["Spear"] = true
}

local enabled = false
local busy = false

local teleportDelay = 0
local stayDuration = 100

-- Position relative to target:
-- +X = target's right
-- +Z = behind target
-- Same orientation as target, so your left side faces them.
local targetOffset = CFrame.new(2, 0, 3)

--==================================================
-- GUI
--==================================================

gui = Instance.new("ScreenGui")
gui.Name = GUI_NAME
gui.ResetOnSpawn = false
gui.DisplayOrder = 999
gui.Parent = playerGui

track(gui.Destroying:Connect(function()
	cleanup()
end))

local frame = Instance.new("Frame")
frame.Size = UDim2.fromOffset(330, 320)
frame.Position = UDim2.new(0.5, -165, 0.5, -160)
frame.BackgroundColor3 = Color3.fromRGB(20, 30, 55)
frame.BorderSizePixel = 0
frame.Active = true
frame.Draggable = true
frame.Parent = gui

local corner = Instance.new("UICorner")
corner.CornerRadius = UDim.new(0, 12)
corner.Parent = frame

local function make(class, props)
	local obj = Instance.new(class)

	for key, value in pairs(props) do
		obj[key] = value
	end

	return obj
end

local toggle = make("TextButton", {
	Parent = frame,
	Size = UDim2.fromOffset(65, 28),

	BackgroundColor3 = Color3.fromRGB(160, 55, 55),
	Text = "OFF",
	TextColor3 = Color3.new(1, 1, 1),
	Font = Enum.Font.GothamBold,
	TextSize = 13
})

local nameBox = make("TextBox", {
	Parent = frame,
	Size = UDim2.new(1, -24, 0, 36),
	Position = UDim2.fromOffset(12, 45),
	BackgroundColor3 = Color3.fromRGB(35, 55, 85),
	PlaceholderText = "Target player name",
	Text = "",
	TextColor3 = Color3.new(1, 1, 1),
	Font = Enum.Font.Gotham,

local function createSlider(y, labelText, initialValue, callback)
	local label = make("TextLabel", {
		Parent = frame,
		Size = UDim2.new(1, -24, 0, 24),
		Position = UDim2.fromOffset(12, y),
		BackgroundTransparency = 1,
		Text = labelText .. ": " .. initialValue .. " ms",
		TextColor3 = Color3.new(1, 1, 1),
		Font = Enum.Font.Gotham,
		TextSize = 13,
		TextXAlignment = Enum.TextXAlignment.Left
	})

	local bar = make("Frame", {
		Parent = frame,
		Size = UDim2.new(1, -40, 0, 10),
		Position = UDim2.fromOffset(20, y + 30),
		BackgroundColor3 = Color3.fromRGB(55, 75, 110),
		BorderSizePixel = 0
	})

	local fill = make("Frame", {
		Parent = bar,
		Size = UDim2.new(0, 0, 1, 0),
		BackgroundColor3 = Color3.fromRGB(75, 145, 255),
		BorderSizePixel = 0
	})

	local knob = make("TextButton", {
		Parent = bar,
		Size = UDim2.fromOffset(18, 18),
		Position = UDim2.new(0, -9, 0.5, -9),
		BackgroundColor3 = Color3.new(1, 1, 1),
		Text = "",
		AutoButtonColor = false
	})

	local dragging = false
	local maxValue = 10000

	local function update(x)
		if not isCurrent() then return end

		local width = math.max(bar.AbsoluteSize.X, 1)
		local ratio = math.clamp(
			(x - bar.AbsolutePosition.X) / width,
			0,
			1
		)

		local value = math.floor(ratio * maxValue)
		fill.Size = UDim2.new(ratio, 0, 1, 0)
		knob.Position = UDim2.new(ratio, -9, 0.5, -9)

		label.Text = labelText .. ": " .. value .. " ms"
		callback(value)
	end

	track(knob.MouseButton1Down:Connect(function()
		dragging = true
	end))

	track(bar.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 then
			dragging = true
			update(input.Position.X)
		end
	end))

	track(UIS.InputChanged:Connect(function(input)
		if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
			update(input.Position.X)
		end
	end))

	track(UIS.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 then
			dragging = false
		end
	end))
end

createSlider(95, "Teleport delay", teleportDelay, function(value)
	teleportDelay = value
end)

createSlider(165, "Stay duration", stayDuration, function(value)
	stayDuration = math.max(1, value)
end)

--==================================================
-- PLAYER / TOOL HELPERS
--==================================================

local function getEquippedTool()
	local character = player.Character
	if not character then return nil end

	for _, item in ipairs(character:GetChildren()) do
		if item:IsA("Tool") and allowedTools[item.Name] then
			return item
		end
	end

	return nil
end

local function findTarget(query)
	query = string.lower(query)

	if query == "" then return nil end

	for _, candidate in ipairs(Players:GetPlayers()) do
		if candidate ~= player
			and string.sub(string.lower(candidate.Name), 1, #query) == query then
			return candidate
		end
	end

	return nil
end

--==================================================
-- TELEPORT CYCLE
--==================================================

local function teleportCycle()
	if not isCurrent() or not enabled or busy then return end
	if not getEquippedTool() then return end

	local query = nameBox.Text
	if query == "" then
		status.Text = "Enter a target name"
		return
	end

	local target = findTarget(query)
	if not target then
		status.Text = "Player not found"
		return
	end

	busy = true
	status.Text = "Waiting " .. teleportDelay .. " ms..."

	local delayAtStart = teleportDelay
	local stayAtStart = stayDuration

	task.wait(delayAtStart / 1000)

	if not isCurrent() then return end

	local character = player.Character
	local targetCharacter = target.Character

	if not character or not targetCharacter then
		status.Text = "Character unavailable"
		busy = false
		return
	end

	local root = character:FindFirstChild("HumanoidRootPart")
	local targetRoot = targetCharacter:FindFirstChild("HumanoidRootPart")

	if not root or not targetRoot then
		status.Text = "Root part unavailable"
		busy = false
		return
	end

	local original = root.CFrame

	root.CFrame = targetRoot.CFrame * targetOffset
	status.Text = "Behind target: " .. target.Name

	task.wait(stayAtStart / 1000)

	if not isCurrent() then return end

	if root.Parent and player.Character == character then
		root.CFrame = original
	end

	status.Text = "Returned to original position"
	busy = false
end

--==================================================
-- INPUT: ONLY ACTIVATES THROUGH ALLOWED TOOLS
--==================================================

local boundTools = {}

local function bindTool(tool)
	if not tool:IsA("Tool") or not allowedTools[tool.Name] then
		return
	end

	if boundTools[tool] then return end
	boundTools[tool] = true

	track(tool.Activated:Connect(function()
		teleportCycle()
	end))
end

local function bindCharacter(character)
	for _, child in ipairs(character:GetChildren()) do
		bindTool(child)
	end

	track(character.ChildAdded:Connect(bindTool))
end

if player.Character then
	bindCharacter(player.Character)
end

track(player.CharacterAdded:Connect(function(character)
	boundTools = {}
	bindCharacter(character)
end))

--==================================================
-- CLEANUP IF THIS SCRIPT IS REPLACED
--==================================================

track(playerGui.ChildRemoved:Connect(function(child)
	if child == gui then
		cleanup()
	end
end))
