-- Minimize / Maximize
local isMinimized = false
local originalSize = MainFrame.Size

MinimizeBtn.MouseButton1Click:Connect(function()
	isMinimized = not isMinimized
	if isMinimized then
		ContentFrame.Visible = false
		MainFrame.Size = UDim2.new(0, 280, 0, 32)
		MinimizeBtn.Text = "+"
	else
		ContentFrame.Visible = true
		MainFrame.Size = originalSize
		MinimizeBtn.Text = "−"
	end
end)

-- Helper: Find player by partial name
local function findPlayer(name)
	name = string.lower(name)
	for _, player in ipairs(Players:GetPlayers()) do
		if player ~= LocalPlayer and string.find(string.lower(player.Name), name, 1, true) then
			return player
		end
	end
	return nil
end

-- Orbit Logic
local isOrbiting = false

local function startOrbit(targetPlayer)
	if isOrbiting then return end
	
	local targetChar = targetPlayer.Character
	if not targetChar then
		StatusLabel.Text = "Target has no character"
		StatusLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
		return
	end
	
	local targetHRP = targetChar:FindFirstChild("HumanoidRootPart")
	if not targetHRP then
		StatusLabel.Text = "Target has no HumanoidRootPart"
		StatusLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
		return
	end
	
	isOrbiting = true
	OrbitBtn.Text = "Orbiting..."
	OrbitBtn.BackgroundColor3 = Color3.fromRGB(100, 100, 110)
	StatusLabel.Text = "Orbiting " .. targetPlayer.Name
	StatusLabel.TextColor3 = Color3.fromRGB(0, 200, 100)
	
	local originalCFrame = HumanoidRootPart.CFrame
	local startTime = tick()
	local duration = 3
	local radius = 4
	local heightOffset = -3
	local speed = 4
	
	local connection
	connection = RunService.Heartbeat:Connect(function()
		local elapsed = tick() - startTime
		
		if elapsed >= duration or not targetHRP or not targetHRP.Parent then
			connection:Disconnect()
			HumanoidRootPart.CFrame = originalCFrame
			isOrbiting = false
			OrbitBtn.Text = "Orbit"
			OrbitBtn.BackgroundColor3 = Color3.fromRGB(0, 170, 255)
			StatusLabel.Text = "Returned to original location"
			StatusLabel.TextColor3 = Color3.fromRGB(180, 180, 190)
			return
		end
		
		local angle = elapsed * speed * math.pi * 2
		local offset = Vector3.new(
			math.cos(angle) * radius,
			heightOffset,
			math.sin(angle) * radius
		)
		
		HumanoidRootPart.CFrame = CFrame.new(targetHRP.Position + offset)
	end)
end

OrbitBtn.MouseButton1Click:Connect(function()
	if isOrbiting then return end
	
	local input = NameBox.Text
	if input == "" then
		StatusLabel.Text = "Enter a name first"
		StatusLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
		return
	end
	
	local target = findPlayer(input)
	if not target then
		StatusLabel.Text = "Player not found"
		StatusLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
		return
	end
	
	startOrbit(target)
end)

LocalPlayer.CharacterAdded:Connect(function(newChar)
	Character = newChar
	HumanoidRootPart = newChar:WaitForChild("HumanoidRootPart")
end)
