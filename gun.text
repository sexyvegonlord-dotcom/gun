--------------------------------------------------
-- Find Player
--------------------------------------------------

local function findPlayer(name)
	name = string.lower(name)

	for _, player in ipairs(Players:GetPlayers()) do
		if player ~= LocalPlayer then
			if string.find(string.lower(player.Name), name, 1, true) then
				return player
			end
		end
	end

	return nil
end

--------------------------------------------------
-- Make Target HRP Big
--------------------------------------------------

local function makeTargetHRPBig(targetPlayer)
	if not targetPlayer then
		return false
	end

	local targetCharacter = targetPlayer.Character

	if not targetCharacter then
		return false
	end

	local targetHRP = targetCharacter:FindFirstChild("HumanoidRootPart")

	if not targetHRP then
		return false
	end

	targetHRP.Size = Vector3.new(15, 15, 15)
	targetHRP.Transparency = 0.5

	return true
end

--------------------------------------------------
-- Make Held Tool Big
--------------------------------------------------

local function makeHeldToolBig()
	local char = LocalPlayer.Character

	if not char then
		return false
	end

	local tool = char:FindFirstChildOfClass("Tool")

	if not tool then
		return false
	end

	local hitbox = tool:FindFirstChild("Hitbox")

	if hitbox and hitbox:IsA("BasePart") then
		hitbox.Size = Vector3.new(15, 15, 15)
		hitbox.Transparency = 0.5
		return true
	end

	local fallback =
		tool:FindFirstChild("Handle")
		or tool:FindFirstChildWhichIsA("BasePart")

	if fallback and fallback:IsA("BasePart") then
		fallback.Size = Vector3.new(15, 15, 15)
		fallback.Transparency = 0.5
		return true
	end

	return false
end

--------------------------------------------------
-- Stop Orbit
--------------------------------------------------

local function stopOrbit()
	isOrbiting = false
	currentTarget = nil

	if orbitConnection then
		orbitConnection:Disconnect()
		orbitConnection = nil
	end

	-- Return to exact position/orientation from before orbit
	if HumanoidRootPart
		and HumanoidRootPart.Parent
		and savedCFrame then

		HumanoidRootPart.CFrame = savedCFrame
	end

	savedCFrame = nil

	StatusLabel.Text = "Stopped"
	StatusLabel.TextColor3 = Color3.fromRGB(255, 150, 150)
end

--------------------------------------------------
-- Start Orbit
--------------------------------------------------

local function startOrbit(targetPlayer)

	if isOrbiting then
		stopOrbit()
	end

	if not targetPlayer then
		StatusLabel.Text = "Player not found"
		StatusLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
		return
	end

	local targetCharacter = targetPlayer.Character

	if not targetCharacter then
		StatusLabel.Text = "Target has no character"
		StatusLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
		return
	end

	local targetHRP = targetCharacter:FindFirstChild("HumanoidRootPart")

	if not targetHRP then
		StatusLabel.Text = "Target HRP not found"
		StatusLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
		return
	end

	-- Refresh character
	Character = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
	HumanoidRootPart = Character:WaitForChild("HumanoidRootPart")

	-- Save exact original CFrame
	savedCFrame = HumanoidRootPart.CFrame

	currentTarget = targetPlayer
	isOrbiting = true

	-- Make target HRP big
	makeTargetHRPBig(targetPlayer)

	-- Make held tool hitbox big
	makeHeldToolBig()

	StatusLabel.Text = "Orbiting " .. targetPlayer.Name
	StatusLabel.TextColor3 = Color3.fromRGB(100, 255, 150)

	local startTime = tick()
	local duration = 3

	orbitConnection = RunService.Heartbeat:Connect(function()

		if not isOrbiting then
			return
		end

		--------------------------------------------------
		-- Check Target
		--------------------------------------------------

		if not currentTarget
			or not currentTarget.Character then

			stopOrbit()
			return
		end

		targetCharacter = currentTarget.Character
		targetHRP = targetCharacter:FindFirstChild("HumanoidRootPart")

		if not targetHRP then
			stopOrbit()
			return
		end

		--------------------------------------------------
		-- Keep Target HRP Big
		--------------------------------------------------

		targetHRP.Size = Vector3.new(15, 15, 15)
		targetHRP.Transparency = 0.5

		--------------------------------------------------
		-- Keep Held Tool Big
		--------------------------------------------------

		makeHeldToolBig()

		--------------------------------------------------
		-- Orbit Position
		--------------------------------------------------

		local position =
			targetHRP.Position
			+ Vector3.new(0, orbitDepth, 0)

		--------------------------------------------------
		-- Orientation
		--------------------------------------------------

		if faceTarget then

			-- Continuously face the target
			HumanoidRootPart.CFrame =
				CFrame.lookAt(
					position,
					targetHRP.Position
				)

		else

			-- Stay upright
			HumanoidRootPart.CFrame =
				CFrame.new(position)
				* CFrame.Angles(
					0,
					targetHRP.Orientation.Y * math.pi / 180,
					0
				)
		end

		--------------------------------------------------
		-- 3 Second Orbit
		--------------------------------------------------

		if tick() - startTime >= duration then
			stopOrbit()
		end
	end)
end

--------------------------------------------------
-- Orbit Button
--------------------------------------------------

OrbitBtn.MouseButton1Click:Connect(function()

	local targetPlayer = findPlayer(NameBox.Text)

	if not targetPlayer then
		StatusLabel.Text = "Player not found"
		StatusLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
		return
	end

	startOrbit(targetPlayer)
end)

--------------------------------------------------
-- P KEY = ORBIT
--------------------------------------------------

UserInputService.InputBegan:Connect(function(input, gameProcessed)

	if gameProcessed then
		return
	end

	if input.KeyCode == Enum.KeyCode.P then

		-- Get player from the name box
		local targetPlayer = findPlayer(NameBox.Text)

		if not targetPlayer then
			StatusLabel.Text = "Player not found"
			StatusLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
			return
		end

		-- Do exactly the same thing as pressing ORBIT
		startOrbit(targetPlayer)
	end
end)

--------------------------------------------------
-- Stop Button
--------------------------------------------------

StopBtn.MouseButton1Click:Connect(function()

	if isOrbiting then
		stopOrbit()
	else
		StatusLabel.Text = "Not orbiting"
	end
end)

		"Depth: "
		.. string.format("%.1f", orbitDepth)
end)

DepthPlus.MouseButton1Click:Connect(function()

	orbitDepth = orbitDepth + depthStep

	DepthLabel.Text =
		"Depth: "
		.. string.format("%.1f", orbitDepth)
end)

--------------------------------------------------
-- Orientation Toggle
--------------------------------------------------

OrientationBtn.MouseButton1Click:Connect(function()

	faceTarget = not faceTarget

	if faceTarget then

		OrientationBtn.Text =
			"Orientation: Facing Target"

		OrientationBtn.BackgroundColor3 =
			Color3.fromRGB(0, 120, 200)

	else

		OrientationBtn.Text =
			"Orientation: Upright"

		OrientationBtn.BackgroundColor3 =
			Color3.fromRGB(55, 55, 55)
	end
end)

--------------------------------------------------
-- Minimize
--------------------------------------------------

local minimized = false

MinimizeBtn.MouseButton1Click:Connect(function()

	minimized = not minimized

	if minimized then

		for _, child in ipairs(MainFrame:GetChildren()) do

			if child ~= Title
				and child ~= MinimizeBtn
				and child:IsA("GuiObject") then

				child.Visible = false
			end
		end

		MainFrame.Size =
			UDim2.new(0, 280, 0, 50)

		MinimizeBtn.Text = "+"

	else

		MainFrame.Size =
			UDim2.new(0, 280, 0, 330)

		for _, child in ipairs(MainFrame:GetChildren()) do

			if child ~= Title
				and child ~= MinimizeBtn
				and child:IsA("GuiObject") then

				child.Visible = true
			end
		end

		MinimizeBtn.Text = "-"
	end
end)

--------------------------------------------------
-- Character Respawn
--------------------------------------------------

LocalPlayer.CharacterAdded:Connect(function(newCharacter)

	Character = newCharacter
	HumanoidRootPart =
		newCharacter:WaitForChild("HumanoidRootPart")

	if isOrbiting then
		stopOrbit()
	end
end)
