-- ========== FUNCTIONS ==========

local function spawnFloor()
	if floorPart then floorPart:Destroy() end
	local hrp = HumanoidRootPart
	if not hrp then return end

	local part = Instance.new("Part")
	part.Name = "SuperFloor"
	part.Size = Vector3.new(1000, floorThickness, 1000)
	part.Position = hrp.Position - Vector3.new(0, 30 + (floorThickness / 2), 0)
	part.Anchored = true
	part.CanCollide = true
	part.Material = Enum.Material.SmoothPlastic
	part.Color = Color3.fromRGB(70, 70, 80)
	part.Parent = workspace

	floorPart = part
	StatusLabel.Text = "Floor spawned (1000x1000)"
	StatusLabel.TextColor3 = Color3.fromRGB(0, 200, 100)
end

local function spawnSky()
	if skyPart then skyPart:Destroy() end
	local hrp = HumanoidRootPart
	if not hrp then return end

	local part = Instance.new("Part")
	part.Name = "SkyCrush"
	part.Size = Vector3.new(200, skyThickness, 200)
	part.Position = hrp.Position + Vector3.new(0, 25 + (skyThickness / 2), 0)
	part.Anchored = true
	part.CanCollide = true
	part.Material = Enum.Material.SmoothPlastic
	part.Color = Color3.fromRGB(120, 40, 160)
	part.Parent = workspace

	skyPart = part
	StatusLabel.Text = "Sky crush block spawned"
	StatusLabel.TextColor3 = Color3.fromRGB(200, 100, 255)
end

local function changeFloorThickness(amount)
	floorThickness = math.clamp(floorThickness + amount, 0.1, 80)
	FloorThickLabel.Text = string.format("Floor Thickness: %.1f", floorThickness)
	if floorPart and floorPart.Parent then
		local oldSize = floorPart.Size
		floorPart.Size = Vector3.new(1000, floorThickness, 1000)
		floorPart.Position = floorPart.Position + Vector3.new(0, (floorThickness - oldSize.Y) / 2, 0)
	end
end

local function changeSkyThickness(amount)
	skyThickness = math.clamp(skyThickness + amount, 0.1, 100)
	SkyThickLabel.Text = string.format("Sky Thickness: %.1f", skyThickness)
	if skyPart and skyPart.Parent then
		local oldSize = skyPart.Size
		local topY = skyPart.Position.Y + (oldSize.Y / 2)
		skyPart.Size = Vector3.new(200, skyThickness, 200)
		skyPart.Position = Vector3.new(skyPart.Position.X, topY - (skyThickness / 2), skyPart.Position.Z)
	end
end

local function deleteSky()
	if skyPart then
		skyPart:Destroy()
		skyPart = nil
		StatusLabel.Text = "Sky block deleted"
		StatusLabel.TextColor3 = Color3.fromRGB(255, 100, 100)
	end
end

-- Make ALL parts above see-through + no collide
local function makeAboveSeeThrough()
	affectedParts = {}
	local hrp = HumanoidRootPart
	if not hrp then return end

	local origin = hrp.Position
	local direction = Vector3.new(0, 200, 0)
	local filterList = {Character}

	while true do
		local rayParams = RaycastParams.new()
		rayParams.FilterDescendantsInstances = filterList
		rayParams.FilterType = Enum.RaycastFilterType.Exclude

		local result = workspace:Raycast(origin, direction, rayParams)
		if result and result.Instance and result.Instance:IsA("BasePart") then
			local part = result.Instance
			table.insert(affectedParts, {
				Part = part,
				OriginalTransparency = part.Transparency,
				OriginalCanCollide = part.CanCollide
			})
			part.Transparency = 0.5
			part.CanCollide = false
			table.insert(filterList, part)
		else
			break
		end
	end

	if #affectedParts > 0 then
		StatusLabel.Text = "Made " .. #affectedParts .. " part(s) see-through"
		StatusLabel.TextColor3 = Color3.fromRGB(0, 200, 100)
	else
		StatusLabel.Text = "No parts found above you"
		StatusLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
	end
end

-- Undo ALL
local function undoSeeThrough()
	for _, data in pairs(affectedParts) do
		if data.Part and data.Part.Parent then
			data.Part.Transparency = data.OriginalTransparency
			data.Part.CanCollide = data.OriginalCanCollide
		end
	end
	affectedParts = {}
	StatusLabel.Text = "All parts restored"
	StatusLabel.TextColor3 = Color3.fromRGB(180, 180, 190)
end

-- Tool Big
local function toolBig()
	local char = LocalPlayer.Character
	if not char then return end
	local tool = char:FindFirstChildOfClass("Tool")
	if not tool then
		StatusLabel.Text = "You are not holding a tool"
		StatusLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
		return
	end

	local hitbox = tool:FindFirstChild("Hitbox") or tool:FindFirstChild("Handle") or tool:FindFirstChildWhichIsA("BasePart")
	if hitbox and hitbox:IsA("BasePart") then
		hitbox.Size = Vector3.new(15, 15, 15)
		hitbox.Transparency = 0.5
		StatusLabel.Text = "Tool hitbox made big (15x15x15)"
		StatusLabel.TextColor3 = Color3.fromRGB(0, 220, 150)
	else
		StatusLabel.Text = "No hitbox/handle found in tool"
		StatusLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
	end
end

-- Character Hitbox
local function charHitbox()
	local name = NameBox.Text
	if name == "" then
		StatusLabel.Text = "Enter a player name"
		StatusLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
		return
	end

	local target = nil
	name = string.lower(name)
	for _, plr in pairs(Players:GetPlayers()) do
		if string.find(string.lower(plr.Name), name, 1, true) then
			target = plr
			break
		end
	end

	if not target or not target.Character then
		StatusLabel.Text = "Player not found"
		StatusLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
		return
	end

	local hrp = target.Character:FindFirstChild("HumanoidRootPart")
	if hrp then
		hrp.Size = Vector3.new(15, 15, 15)
		hrp.Transparency = 0.5
		StatusLabel.Text = target.Name .. " HRP made big"
		StatusLabel.TextColor3 = Color3.fromRGB(255, 100, 100)
	else
		StatusLabel.Text = "No HumanoidRootPart found"
		StatusLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
	end
end
