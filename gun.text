	if isMinimized then
		ContentFrame.Visible = false

		ContentFrame.Visible = true

-- FIND PLAYER
local function findPlayer(name)
	name = string.lower(name)

	for _, player in ipairs(Players:GetPlayers()) do
		if player ~= LocalPlayer then
			if string.find(string.lower(player.Name), name, 1, true) then
				return player
			end
		end
	end

	return nil
end

-- UPDATE DEPTH DISPLAY
local function updateDepth()
	DepthLabel.Text = string.format("Depth: %.1f", orbitDepth)
end

-- DEPTH CONTROLS
DepthMinus.MouseButton1Click:Connect(function()
	orbitDepth -= depthStep
	orbitDepth = math.round(orbitDepth * 10) / 10
	updateDepth()
end)

DepthPlus.MouseButton1Click:Connect(function()
	orbitDepth += depthStep
	orbitDepth = math.round(orbitDepth * 10) / 10
	updateDepth()
end)

-- ORIENTATION
OrientationBtn.MouseButton1Click:Connect(function()
	faceTarget = not faceTarget

	if faceTarget then
		OrientationBtn.Text = "Orientation: Facing Target"
		OrientationBtn.BackgroundColor3 = Color3.fromRGB(0, 150, 100)
	else
		OrientationBtn.Text = "Orientation: Upright"
		OrientationBtn.BackgroundColor3 = Color3.fromRGB(70, 70, 80)
	end
end)

-- ORBIT
local isOrbiting = false
local orbitConnection = nil
local savedCFrame = nil

local function stopOrbit()
	if orbitConnection then
		orbitConnection:Disconnect()
		orbitConnection = nil
	end

	if savedCFrame and HumanoidRootPart then
		HumanoidRootPart.CFrame = savedCFrame
	end

	isOrbiting = false

	OrbitBtn.Text = "Orbit"
	OrbitBtn.BackgroundColor3 = Color3.fromRGB(0, 170, 255)

	StatusLabel.Text = "Stopped / Returned"
	StatusLabel.TextColor3 = Color3.fromRGB(180, 180, 190)
end

local function startOrbit(targetPlayer)
	if isOrbiting then
		return
	end

	local targetChar = targetPlayer.Character

	if not targetChar then
		StatusLabel.Text = "Target has no character"
		StatusLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
		return
	end

	local targetHRP = targetChar:FindFirstChild("HumanoidRootPart")

	if not targetHRP then
		StatusLabel.Text = "Target has no HumanoidRootPart"
		StatusLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
		return
	end

	isOrbiting = true

	OrbitBtn.Text = "Underneath..."
	OrbitBtn.BackgroundColor3 = Color3.fromRGB(100, 100, 110)

	StatusLabel.Text = "Under " .. targetPlayer.Name
	StatusLabel.TextColor3 = Color3.fromRGB(0, 200, 100)

	savedCFrame = HumanoidRootPart.CFrame

	local startTime = tick()
	local duration = 3

	orbitConnection = RunService.Heartbeat:Connect(function()
		local elapsed = tick() - startTime

		if elapsed >= duration
			or not targetHRP
			or not targetHRP.Parent
			or not HumanoidRootPart
			or not HumanoidRootPart.Parent then

			stopOrbit()
			return
		end

		-- Position underneath target
		local position = targetHRP.Position + Vector3.new(0, orbitDepth, 0)

		if faceTarget then
			-- Tilt upward so your character faces the target above
			HumanoidRootPart.CFrame = CFrame.lookAt(
				position,
				targetHRP.Position
			)
		else
			-- Stay vertically upright
			HumanoidRootPart.CFrame =
				CFrame.new(position) *
				CFrame.Angles(0, targetHRP.Orientation.Y * math.pi / 180, 0)
		end
	end)
end

-- ORBIT BUTTON
OrbitBtn.MouseButton1Click:Connect(function()
	if isOrbiting then
		return
	end

	local input = NameBox.Text

	if input == "" then
		StatusLabel.Text = "Enter a name first"
		StatusLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
		return
	end

	local target = findPlayer(input)

	if not target then
		StatusLabel.Text = "Player not found"
		StatusLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
		return
	end

	startOrbit(target)
end)

-- STOP
StopBtn.MouseButton1Click:Connect(function()
	if isOrbiting then
		stopOrbit()
	end
end)

-- CHARACTER RESPAWN
LocalPlayer.CharacterAdded:Connect(function(newChar)
	Character = newChar
	HumanoidRootPart = newChar:WaitForChild("HumanoidRootPart")
end)
