--------------------------------------------------
-- FIND PLAYER
--------------------------------------------------

local function findPlayer(name)

	name = string.lower(name)

	if name == "" then
		return nil
	end

	for _, player in ipairs(Players:GetPlayers()) do

		if player ~= LocalPlayer then

			if string.find(
				string.lower(player.Name),
				name,
				1,
				true
			) then

				return player
			end
		end
	end

	return nil
end

--------------------------------------------------
-- MAKE TARGET HRP BIG
--------------------------------------------------

local function makeTargetHRPBig(targetPlayer)

	if not targetPlayer then
		return false
	end

	local targetCharacter =
		targetPlayer.Character

	if not targetCharacter then
		return false
	end

	local targetHRP =
		targetCharacter:FindFirstChild(
			"HumanoidRootPart"
		)

	if not targetHRP then
		return false
	end

	targetHRP.Size =
		Vector3.new(15, 15, 15)

	targetHRP.Transparency = 0.5

	return true
end

--------------------------------------------------
-- MAKE HELD TOOL BIG
--------------------------------------------------

local function makeHeldToolBig()

	local char =
		LocalPlayer.Character

	if not char then
		return false
	end

	local tool =
		char:FindFirstChildOfClass("Tool")

	if not tool then
		return false
	end

	--------------------------------------------------
	-- ACTUAL HITBOX
	--------------------------------------------------

	local hitbox =
		tool:FindFirstChild("Hitbox")

	if hitbox and hitbox:IsA("BasePart") then

		hitbox.Size =
			Vector3.new(15, 15, 15)

		hitbox.Transparency = 0.5

		return true
	end

	--------------------------------------------------
	-- FALLBACK
	--------------------------------------------------

	local fallback =
		tool:FindFirstChild("Handle")
		or tool:FindFirstChildWhichIsA("BasePart")

	if fallback and fallback:IsA("BasePart") then

		fallback.Size =
			Vector3.new(15, 15, 15)

		fallback.Transparency = 0.5

		return true
	end

	return false
end

--------------------------------------------------
-- STOP ORBIT
--------------------------------------------------

local function stopOrbit()

	isOrbiting = false
	currentTarget = nil

	--------------------------------------------------
	-- STOP LOOP FIRST
	--------------------------------------------------

	if orbitConnection then

		orbitConnection:Disconnect()

		orbitConnection = nil
	end

	--------------------------------------------------
	-- RETURN TO EXACT STARTING CFRAME
	--------------------------------------------------

	if
		savedCFrame
		and HumanoidRootPart
		and HumanoidRootPart.Parent
	then

		-- Restore the EXACT CFrame from
		-- immediately before the first P
		HumanoidRootPart.CFrame = savedCFrame
	end

	--------------------------------------------------
	-- CLEAR SAVED POSITION AFTER RESTORING
	--------------------------------------------------

	savedCFrame = nil

	StatusLabel.Text = "Returned to starting position"

	StatusLabel.TextColor3 =
		Color3.fromRGB(100, 255, 150)
end

--------------------------------------------------
-- START ORBIT
--------------------------------------------------

local function startOrbit(targetPlayer)

	--------------------------------------------------
	-- VALIDATE TARGET
	--------------------------------------------------

	if not targetPlayer then

		StatusLabel.Text =
			"Player not found"

		StatusLabel.TextColor3 =
			Color3.fromRGB(255, 80, 80)

		return
	end

	local targetCharacter =
		targetPlayer.Character

	if not targetCharacter then

		StatusLabel.Text =
			"Target has no character"

		StatusLabel.TextColor3 =
			Color3.fromRGB(255, 80, 80)

		return
	end

	local targetHRP =
		targetCharacter:FindFirstChild(
			"HumanoidRootPart"
		)

	if not targetHRP then

		StatusLabel.Text =
			"Target HRP not found"

		StatusLabel.TextColor3 =
			Color3.fromRGB(255, 80, 80)

		return
	end

	--------------------------------------------------
	-- REFRESH OWN CHARACTER
	--------------------------------------------------

	Character =
		LocalPlayer.Character
		or LocalPlayer.CharacterAdded:Wait()

	HumanoidRootPart =
		Character:WaitForChild(
			"HumanoidRootPart"
		)

	--------------------------------------------------
	-- SAVE POSITION BEFORE DOING ANYTHING
	--------------------------------------------------

	-- This is captured ONCE when P starts orbit.
	-- It will not change while orbiting.

	savedCFrame =
		HumanoidRootPart.CFrame

	--------------------------------------------------
	-- SAVE TARGET
	--------------------------------------------------

	currentTarget = targetPlayer

	isOrbiting = true

	--------------------------------------------------
	-- MAKE TARGET + TOOL BIG
	--------------------------------------------------

	makeTargetHRPBig(targetPlayer)
	makeHeldToolBig()

	StatusLabel.Text =
		"Orbiting " .. targetPlayer.Name

	StatusLabel.TextColor3 =
		Color3.fromRGB(100, 255, 150)

	--------------------------------------------------
	-- ORBIT LOOP
	--------------------------------------------------

	orbitConnection =
		RunService.Heartbeat:Connect(function()

			if not isOrbiting then
				return
			end

			--------------------------------------------------
			-- GET TARGET AGAIN
			--------------------------------------------------

			if
				not currentTarget
				or not currentTarget.Character
			then

				stopOrbit()
				return
			end

			targetCharacter =
				currentTarget.Character

			targetHRP =
				targetCharacter:FindFirstChild(
					"HumanoidRootPart"
				)

			if not targetHRP then

				stopOrbit()
				return
			end

			--------------------------------------------------
			-- KEEP TARGET HRP BIG
			--------------------------------------------------

			targetHRP.Size =
				Vector3.new(15, 15, 15)

			targetHRP.Transparency = 0.5

			--------------------------------------------------
			-- KEEP TOOL BIG
			--------------------------------------------------

			makeHeldToolBig()

			--------------------------------------------------
			-- POSITION UNDER TARGET
			--------------------------------------------------

			local position =
				targetHRP.Position
				+ Vector3.new(
					0,
					orbitDepth,
					0
				)

			--------------------------------------------------
			-- FACING TARGET
			--------------------------------------------------

			if faceTarget then

				HumanoidRootPart.CFrame =
					CFrame.lookAt(
						position,
						targetHRP.Position
					)

			else

				--------------------------------------------------
				-- UPRIGHT
				--------------------------------------------------

				HumanoidRootPart.CFrame =
					CFrame.new(position)
					*
					CFrame.Angles(
						0,
						targetHRP.Orientation.Y
							* math.pi / 180,
						0
					)
			end
		end)
end

--------------------------------------------------
-- ORBIT BUTTON
--------------------------------------------------

OrbitBtn.MouseButton1Click:Connect(function()

	--------------------------------------------------
	-- IF ORBITING = STOP + RETURN
	--------------------------------------------------

	if isOrbiting then

		stopOrbit()
		return
	end

	--------------------------------------------------
	-- START
	--------------------------------------------------

	local targetPlayer =
		findPlayer(NameBox.Text)

	if not targetPlayer then

		StatusLabel.Text =
			"Player not found"

		StatusLabel.TextColor3 =
			Color3.fromRGB(255, 80, 80)

		return
	end

	startOrbit(targetPlayer)
end)

--------------------------------------------------
-- P KEY
--------------------------------------------------

UserInputService.InputBegan:Connect(function(input)

	if input.KeyCode ~= Enum.KeyCode.P then
		return
	end

	--------------------------------------------------
	-- P AGAIN = STOP + RETURN TO EXACT START
	--------------------------------------------------

	if isOrbiting then

		stopOrbit()
		return
	end

	--------------------------------------------------
	-- FIRST P = START
	--------------------------------------------------

	local targetPlayer =
		findPlayer(NameBox.Text)

	if not targetPlayer then

		StatusLabel.Text =
			"Player not found"

		StatusLabel.TextColor3 =
			Color3.fromRGB(255, 80, 80)

		return
	end

	startOrbit(targetPlayer)
end)

--------------------------------------------------
-- STOP BUTTON
--------------------------------------------------

StopBtn.MouseButton1Click:Connect(function()

	if isOrbiting then

		stopOrbit()

	else

		StatusLabel.Text =
			"Not orbiting"
	end
end)

--------------------------------------------------
-- MINIMIZE
--------------------------------------------------

local minimized = false

MinimizeBtn.MouseButton1Click:Connect(function()

	minimized =
		not minimized

	if minimized then

		for _, child in ipairs(
			MainFrame:GetChildren()
		) do

			if
				child ~= Title
				and child ~= MinimizeBtn
				and child:IsA("GuiObject")
			then

				child.Visible = false
			end
		end

		MainFrame.Size =
			UDim2.new(0, 280, 0, 50)

		MinimizeBtn.Text = "+"

	else

		MainFrame.Size =
			UDim2.new(0, 280, 0, 330)

		for _, child in ipairs(
			MainFrame:GetChildren()
		) do

			if
				child ~= Title
				and child ~= MinimizeBtn
				and child:IsA("GuiObject")
			then

				child.Visible = true
			end
		end

		MinimizeBtn.Text = "-"
	end
end)

--------------------------------------------------
-- CHARACTER RESPAWN
--------------------------------------------------

LocalPlayer.CharacterAdded:Connect(function(newCharacter)

	Character = newCharacter

	HumanoidRootPart =
		newCharacter:WaitForChild(
			"HumanoidRootPart"
		)

	--------------------------------------------------
	-- If you die/respawn during orbit,
	-- stop the orbit normally.
	--------------------------------------------------

	if isOrbiting then
		stopOrbit()
	end
end)
