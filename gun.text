
-- LOCAL TELEPORT TESTER
-- Standalone LocalScript for Roblox Studio

local Players = game:GetService("Players")
local UserInputService = game:GetService("UserInputService")

local LocalPlayer = Players.LocalPlayer
local PlayerGui = LocalPlayer:WaitForChild("PlayerGui")

-- Remove previous version
local previous = PlayerGui:FindFirstChild("LocalTeleportTester")
if previous then
    previous:Destroy()
end

local gui = Instance.new("ScreenGui")
gui.Name = "LocalTeleportTester"
gui.ResetOnSpawn = false
gui.DisplayOrder = 999999
gui.Parent = PlayerGui

local frame = Instance.new("Frame")
frame.Size = UDim2.fromOffset(300, 225)
frame.Position = UDim2.new(0.5, -150, 0.5, -112)
frame.BackgroundColor3 = Color3.fromRGB(25, 27, 35)
frame.BorderSizePixel = 0
frame.Active = true
frame.Draggable = true
frame.Parent = gui

local corner = Instance.new("UICorner")
corner.CornerRadius = UDim.new(0, 12)
corner.Parent = frame

local function create(class, props)
    local obj = Instance.new(class)
    for key, value in pairs(props) do
        obj[key] = value
    end
    return obj
end

local title = create("TextLabel", {
    Parent = frame,
    Size = UDim2.new(1, -50, 0, 40),
    Position = UDim2.fromOffset(12, 0),
    BackgroundTransparency = 1,
    Text = "LOCAL TELEPORT TESTER",
    TextColor3 = Color3.new(1,1,1),
    Font = Enum.Font.GothamBold,
    TextSize = 14,
    TextXAlignment = Enum.TextXAlignment.Left
})

local minimize = create("TextButton", {
    Parent = frame,
    Size = UDim2.fromOffset(32, 28),
    Position = UDim2.new(1, -40, 0, 6),
    Text = "—",
    TextColor3 = Color3.new(1,1,1),
    BackgroundColor3 = Color3.fromRGB(55,58,70)
})

local nameBox = create("TextBox", {
    Parent = frame,
    Size = UDim2.new(1, -24, 0, 38),
    Position = UDim2.fromOffset(12, 50),
    PlaceholderText = "Player name or first letters",
    Text = "",
    ClearTextOnFocus = false,
    BackgroundColor3 = Color3.fromRGB(43,46,58),
    TextColor3 = Color3.new(1,1,1),
    Font = Enum.Font.Gotham,
    TextSize = 13
})

local durationLabel = create("TextLabel", {
    Parent = frame,
    Size = UDim2.new(1, -24, 0, 25),
    Position = UDim2.fromOffset(12, 98),
    BackgroundTransparency = 1,
    Text = "Stay duration: 1.0 seconds",
    TextColor3 = Color3.new(1,1,1),
    Font = Enum.Font.Gotham,
    TextSize = 13
})

local duration = 1

local slider = create("TextBox", {
    Parent = frame,
    Size = UDim2.new(1, -24, 0, 30),
    Position = UDim2.fromOffset(12, 128),
    BackgroundColor3 = Color3.fromRGB(43,46,58),
    Text = "1.0",
    TextColor3 = Color3.new(1,1,1),
    Font = Enum.Font.Gotham,
    TextSize = 13
})

slider.FocusLost:Connect(function()
    duration = math.clamp(tonumber(slider.Text) or 1, 0.1, 10)
    slider.Text = tostring(duration)
    durationLabel.Text = "Stay duration: " .. duration .. " seconds"
end)

local run = create("TextButton", {
    Parent = frame,
    Size = UDim2.new(1, -24, 0, 38),
    Position = UDim2.fromOffset(12, 166),
    BackgroundColor3 = Color3.fromRGB(85,110,230),
    Text = "TELEPORT",
    TextColor3 = Color3.new(1,1,1),
    Font = Enum.Font.GothamBold,
    TextSize = 14
})

local status = create("TextLabel", {
    Parent = frame,
    Size = UDim2.new(1, -24, 0, 20),
    Position = UDim2.fromOffset(12, 205),
    BackgroundTransparency = 1,
    Text = "Ready",
    TextColor3 = Color3.fromRGB(170,180,200),
    Font = Enum.Font.Gotham,
    TextSize = 12
})

local busy = false

run.MouseButton1Click:Connect(function()
    if busy then return end

    local query = string.lower(nameBox.Text)
    if query == "" then
        status.Text = "Enter a player name"
        return
    end

    local target

    for _, candidate in ipairs(Players:GetPlayers()) do
        if candidate ~= LocalPlayer
            and string.sub(string.lower(candidate.Name), 1, #query) == query then
            target = candidate
            break
        end
    end

    if not target then
        status.Text = "Player not found"
        return
    end

    local character = LocalPlayer.Character
    local targetCharacter = target.Character

    if not character or not targetCharacter then return end

    local root = character:FindFirstChild("HumanoidRootPart")
    local targetRoot = targetCharacter:FindFirstChild("HumanoidRootPart")

    if not root or not targetRoot then return end

    busy = true

    local original = root.CFrame

    root.CFrame = targetRoot.CFrame * CFrame.new(0, 0, -3)
    status.Text = "At " .. target.Name

    task.wait(duration)

    if root.Parent then
        root.CFrame = original
    end

    status.Text = "Returned"
    busy = false
end)

local minimized = false

minimize.MouseButton1Click:Connect(function()
    minimized = not minimized

    for _, obj in ipairs(frame:GetChildren()) do
        if obj ~= title and obj ~= minimize and obj:IsA("GuiObject") then
            obj.Visible = not minimized
        end
    end

    frame.Size = minimized
        and UDim2.fromOffset(190, 42)
        or UDim2.fromOffset(300, 235)
end)

-- Recreate GUI if another local script removes it
task.spawn(function()
    while task.wait(1) do
        if not gui.Parent then
            break
        end
    end
end)
